cmake_minimum_required(VERSION 3.24)
project(ProcessLassoNoNag LANGUAGES CXX ASM_MASM RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MSVC x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/WX> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
set(NONAG_BUILD "development" CACHE STRING "Release tag or source revision")
set(NONAG_FILE_VERSION "0,0,0,0")
if(NONAG_BUILD MATCHES "^v([0-9]+)\\.([0-9]+)\\.([0-9]+)")
  set(NONAG_FILE_VERSION "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},0")
endif()
configure_file(native/version.rc.in version.rc @ONLY)
add_library(version SHARED native/proxy.cpp native/exports.asm native/version.def native/dialog_hook.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_link_libraries(version PRIVATE user32)
target_link_options(version PRIVATE /DYNAMICBASE /NXCOMPAT /Brepro)
set_target_properties(version PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests/native)
endif()
